packages = [] from js import document, fetch import datetime, asyncio hid = document.getElementById('hotel_id').value start_el = document.getElementById('start') end_el = document.getElementById('end') status_el = document.getElementById('status') table = document.getElementById('grid-table') # Defaults: next 14 days try: today = datetime.date.today() start_el.value = today.isoformat() end_el.value = (today + datetime.timedelta(days=14)).isoformat() except Exception: pass def set_status(msg, kind='error'): status_el.className = f'status show {kind}' status_el.textContent = msg def clear_status(): status_el.className = 'status' status_el.textContent = '' async def load_grid(evt=None): if evt: evt.preventDefault() clear_status() s = start_el.value e = end_el.value if not s or not e: set_status('Select start and end dates') return try: resp = await fetch(f"/api/availability/rooms-grid?hotel_id={hid}&start={s}&end={e}") data = await resp.json() py = data.to_py() if not py.get('ok'): set_status(py.get('error','Failed to load grid')) return dates = py.get('dates') or [] rows = py.get('rows') or [] # Render table table.innerHTML = '' # Header thead = document.createElement('thead') trh = document.createElement('tr') th_room = document.createElement('th') th_room.textContent = 'Room' trh.appendChild(th_room) for d in dates: th = document.createElement('th') th.textContent = d[5:] # MM-DD trh.appendChild(th) thead.appendChild(trh) table.appendChild(thead) # Body tbody = document.createElement('tbody') for r in rows: tr = document.createElement('tr') td0 = document.createElement('td') rn = r.get('room_number') or '' rt = r.get('type') or '' td0.textContent = f"{rn} {('• ' + rt) if rt else ''}" tr.appendChild(td0) cells = r.get('cells') or [] for booked in cells: td = document.createElement('td') td.style.textAlign = 'center' td.style.fontSize = '.75rem' td.textContent = '•' if booked: td.style.background = 'rgba(239,68,68,.18)' td.style.color = '#fca5a5' td.title = 'Booked' else: td.style.background = 'rgba(16,185,129,.12)' td.style.color = '#6ee7b7' td.title = 'Available' tr.appendChild(td) tbody.appendChild(tr) table.appendChild(tbody) except Exception as e: set_status(str(e)) form = document.getElementById('range-form') form.addEventListener('submit', load_grid) asyncio.ensure_future(load_grid()) # Footer year try: document.getElementById('year').textContent = str(datetime.datetime.now().year) except Exception: pass
